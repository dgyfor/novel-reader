        editorText = EditText(this).apply {

        readerContainer = HorizontalScrollView(this).apply {
            visibility = View.GONE
            isHorizontalScrollBarEnabled = false
            isSmoothScrollingEnabled = false
        }
        pageContainer = LinearLayout(this).apply {
            orientation = LinearLayout.HORIZONTAL
        }
        readerContainer?.addView(pageContainer)
        rootLayout?.addView(readerContainer, FrameLayout.LayoutParams(-1, -1))

        readerContainer?.setOnTouchListener(object : View.OnTouchListener {
            private var startX = 0f
            private var startY = 0f
            private var isMoving = false

            override fun onTouch(v: View, event: MotionEvent): Boolean {
                when (event.action) {
                    MotionEvent.ACTION_DOWN -> {
                        startX = event.x
                        startY = event.y
                        isMoving = false
                        return false
                    }
                    MotionEvent.ACTION_MOVE -> {
                        if (abs(event.x - startX) > 40 || abs(event.y - startY) > 40) {
                            isMoving = true
                        }
                        return false
                    }
                    MotionEvent.ACTION_UP -> {
                        if (!isMoving) {
                            val w = v.width
                            if (startX < w / 3f) jumpToPrevPage()
                            else if (startX > w * 2 / 3f) jumpToNextPage()
                            else toggleMenu()
                            v.performClick() // 娑堥櫎鏃犻殰纰嶈鍛?                            return true
                        } else {
                            if (startX - event.x > 100) {
                                jumpToNextPage()
                                return true
                            } else if (event.x - startX > 100) {
                                jumpToPrevPage()
                                return true
                            }
                        }
                        return false
                    }
                }
                return false
            }
        })

        // 3. 椤堕儴鏍?        topBar = createTopBar(book)
        rootLayout?.addView(topBar, FrameLayout.LayoutParams(-1, -2).apply { gravity = Gravity.TOP })

        // 4. 搴曢儴鏍?        bottomBar = createBottomBar()
        rootLayout?.addView(bottomBar, FrameLayout.LayoutParams(-1, -2).apply { gravity = Gravity.BOTTOM })

        // 鍔犺浇鍐呭

        editorText?.setText(content)
        renderHighlights()
        editorText?.setSelection(editorText!!.text.length)
        updateWordCount(content)

        // 闃叉姈淇濆瓨

            override fun onTextChanged(s: CharSequence?, start: Int, before: Int, count: Int) {}

                if (isRenderingSpan || isReaderMode) return
                val text = s?.toString() ?: ""
                dirty = true
                scheduleSave(text)
                updateWordCount(text)
            }
        })

        setContentView(rootLayout)
    }

    private fun toggleMenu() {
        val tb = topBar ?: return
        val bb = bottomBar ?: return
        val isVisible = tb.visibility == View.VISIBLE
        tb.visibility = if (isVisible) View.GONE else View.VISIBLE
        bb.visibility = if (isVisible) View.GONE else View.VISIBLE
    }

    // ================= 鏍稿績锛氱炕椤甸€昏緫 =================
    private fun jumpToNextPage() {
        val pc = pageContainer ?: return
        if (currentPageIndex < pc.childCount - 1) {
            currentPageIndex++
            val w = readerContainer?.width ?: 0
            readerContainer?.scrollTo(currentPageIndex * w, 0)
        } else {
            toast("宸茬粡鏄渶鍚庝竴椤典簡")
        }
    }

    private fun jumpToPrevPage() {
        if (currentPageIndex > 0) {
            currentPageIndex--
            val w = readerContainer?.width ?: 0
            readerContainer?.scrollTo(currentPageIndex * w, 0)
        } else {
            toast("宸茬粡鏄涓€椤典簡")
        }
    }

    // ================= 妯″紡鍒囨崲 =================
    private fun toggleReaderMode() {
        if (isReaderMode) {
            isReaderMode = false
            readerContainer?.visibility = View.GONE
            editorText?.visibility = View.VISIBLE
            btnToggleMode?.text = "闃呰妯″紡"

            val pc = pageContainer
            if (pc != null && currentPageIndex in 0 until pc.childCount) {
                val tv = pc.getChildAt(currentPageIndex) as TextView
                val offset = tv.tag as? Int ?: 0
                editorText?.setSelection(offset.coerceIn(0, editorText?.text?.length ?: 0))
            }

            editorText?.requestFocus()
            val imm = getSystemService(Context.INPUT_METHOD_SERVICE) as InputMethodManager
            imm.showSoftInput(editorText, InputMethodManager.SHOW_IMPLICIT)
        } else {
            isReaderMode = true
            flushNow()
            editorText?.clearFocus()
            val imm = getSystemService(Context.INPUT_METHOD_SERVICE) as InputMethodManager
            imm.hideSoftInputFromWindow(editorText?.windowToken, 0)

            editorText?.visibility = View.GONE
            readerContainer?.visibility = View.VISIBLE
            btnToggleMode?.text = "缂栬緫妯″紡"

            renderPages()
        }
    }

    private fun renderPages() {
        val et = editorText ?: return
        val pc = pageContainer ?: return
        val text = et.text.toString()
        pc.removeAllViews()
        if (text.isEmpty()) return

        et.post {
            if (!isReaderMode) return@post

            val paint: TextPaint = et.paint
            val width = et.width - et.paddingLeft - et.paddingRight
            val height = et.height - et.paddingTop - et.paddingBottom
            if (width <= 0 || height <= 0) return@post

            val staticLayout = StaticLayout.Builder.obtain(text, 0, text.length, paint, width)
                .setLineSpacing(0f, lineSpacing).build()

            val lineHeight = staticLayout.getLineBottom(0) - staticLayout.getLineTop(0)
            if (lineHeight <= 0) return@post
            val linesPerPage = height / lineHeight
            if (linesPerPage <= 0) return@post

            val totalLines = staticLayout.lineCount
            val bookId = book?.id ?: return@post
            val anns = annDao.getAnnotations(bookId)

            var startLine = 0
            while (startLine < totalLines) {
                val endLine = minOf(startLine + linesPerPage, totalLines)
                val startOffset = staticLayout.getLineStart(startLine)
                val endOffset = staticLayout.getLineEnd(endLine - 1)

                val pageText = text.substring(startOffset, endOffset)
                val spannable = SpannableString(pageText)

                for (ann in anns) {
                    val s = maxOf(ann.startOffset, startOffset) - startOffset
                    val e = minOf(ann.endOffset, endOffset) - startOffset
                    if (s < e) {
                        val color = if (ann.note.isNullOrEmpty()) Color.parseColor("#66FFEB3B") else Color.parseColor("#6600E676")
                        spannable.setSpan(BackgroundColorSpan(color), s, e, Spannable.SPAN_EXCLUSIVE_EXCLUSIVE)
                    }
                }

                val tv = TextView(this).apply {
                    setText(spannable) // 馃憟 淇浜?val 鎶ラ敊锛岃繖閲屽繀椤绘槸 setText
                    setTextColor(et.currentTextColor)
                    textSize = fontSize
                    setLineSpacing(0f, lineSpacing)
                    typeface = et.typeface
                    setPadding(et.paddingLeft, et.paddingTop, et.paddingRight, et.paddingBottom)
                    setTextIsSelectable(true)
                    setBackgroundColor(Color.TRANSPARENT)
                    setTag(startOffset)
                }

                setupSelectionMenu(tv, startOffset)
                val lp = LinearLayout.LayoutParams(et.width, ViewGroup.LayoutParams.MATCH_PARENT)
                pc.addView(tv, lp)
                startLine = endLine
            }

            val cursorPos = et.selectionStart
            for (i in 0 until pc.childCount) {
                val tv = pc.getChildAt(i) as TextView
                val pageStartOffset = tv.tag as? Int ?: 0
                val pageEndOffset = pageStartOffset + tv.text.length
                if (cursorPos in pageStartOffset..pageEndOffset) {
                    currentPageIndex = i
                    break
                }
            }
            readerContainer?.scrollTo(currentPageIndex * (readerContainer?.width ?: 0), 0)
        }
    }

    private fun setupSelectionMenu(view: View, pageStartOffset: Int) {
        val callback = object : ActionMode.Callback {
            override fun onCreateActionMode(mode: ActionMode, menu: Menu): Boolean {
                menu.add(0, 1, 0, "鍒掗噸鐐?).setShowAsAction(MenuItem.SHOW_AS_ACTION_ALWAYS)
                menu.add(0, 2, 1, "鍐欐兂娉?).setShowAsAction(MenuItem.SHOW_AS_ACTION_ALWAYS)
                return true
            }

            override fun onPrepareActionMode(mode: ActionMode, menu: Menu) = false

            override fun onActionItemClicked(mode: ActionMode, item: MenuItem): Boolean {
                val start = when (view) {
                    is EditText -> view.selectionStart
                    is TextView -> view.selectionStart + pageStartOffset
                    else -> return false
                }
                val end = when (view) {
                    is EditText -> view.selectionEnd
                    is TextView -> view.selectionEnd + pageStartOffset
                    else -> return false
                }
                if (start < 0 || end <= start) return false

                val fullText = editorText?.text?.toString() ?: return false
                if (end > fullText.length) return false
                val quoted = fullText.substring(start, end)
                val b = book ?: return false

                return when (item.itemId) {
                    1 -> {
                        annDao.insert(b.id, start, end, quoted, null, "yellow")
                        mode.finish()
                        refreshUI()
                        toast("宸叉坊鍔犲垝绾?)
                        true
                    }
                    2 -> {
                        val input = EditText(this@EditorActivity)
                        input.hint = "鍐欎笅浣犵殑鎯虫硶..."
                        input.setPadding(40, 30, 40, 30)
                        AlertDialog.Builder(this@EditorActivity)
                            .setTitle("鎯虫硶").setMessage("鍘熸枃锛?quoted").setView(input)
                            .setPositiveButton("淇濆瓨") { _, _ ->
                                annDao.insert(b.id, start, end, quoted, input.text.toString().trim(), "green")
                                mode.finish()
                                refreshUI()
                                toast("鎯虫硶宸蹭繚瀛?)
                            }.setNegativeButton("鍙栨秷", null).show()
                        true
                    }
                    else -> false
                }
            }
            override fun onDestroyActionMode(mode: ActionMode) {}
        }

        // 淇浜嗗啑浣欑殑绫诲瀷鍒ゆ柇璀﹀憡
        if (view is TextView) view.customSelectionActionModeCallback = callback
    }

    private fun refreshUI() {
        renderHighlights()

    private fun renderHighlights() {
        val et = editorText ?: return
        val b = book ?: return
        isRenderingSpan = true
        val editable = et.text

        for (span in editable.getSpans(0, editable.length, BackgroundColorSpan::class.java)) {
            editable.removeSpan(span)
        }

        for (ann in annDao.getAnnotations(b.id)) {
            // 淇敼涓鸿寖鍥存鏌ワ紝娑堥櫎璀﹀憡
            if (ann.startOffset in 0 until ann.endOffset && ann.endOffset <= editable.length) {
                val color = if (ann.note.isNullOrEmpty()) Color.parseColor("#66FFEB3B") else Color.parseColor("#6600E676")
                editable.setSpan(BackgroundColorSpan(color), ann.startOffset, ann.endOffset, Spannable.SPAN_EXCLUSIVE_EXCLUSIVE)
            }
        }
        isRenderingSpan = false
    }

    private fun textFile(bookId: String) = File(File(filesDir, "books"), "$bookId.txt")
    private fun toast(msg: String) = Toast.makeText(this, msg, Toast.LENGTH_SHORT).show()

    private fun loadContent(book: Book): String {
        val f = textFile(book.id)
        if (f.exists()) return f.readText(Charsets.UTF_8)
        f.parentFile?.mkdirs()
        val legacyFile = File(filesDir, "novel_data.json")
        var content = ""
        if (legacyFile.exists()) {
            try {
                val legacyArr = JSONArray(legacyFile.readText(Charsets.UTF_8))
                for (i in 0 until legacyArr.length()) {
                    val o = legacyArr.optJSONObject(i) ?: continue
                    if (o.optString("id") == book.id) { content = o.optString("content"); break }
                }
            } catch (_: Exception) {}
        }
        f.writeText(content, Charsets.UTF_8)
        return content
    }

    private fun scheduleSave(content: String) {
        saveRunnable?.let { mainHandler.removeCallbacks(it) }
        val b = book ?: return
        saveRunnable = Runnable {
            ioExecutor.execute {
                writeContent(b.id, content)
                mainHandler.post { if (editorText?.text?.toString() == content) dirty = false }
            }
        }
        mainHandler.postDelayed(saveRunnable!!, 350)
    }

    private fun writeContent(bookId: String, content: String) {
        val f = textFile(bookId)
        f.parentFile?.mkdirs()
        val tmp = File(f.parentFile, "${f.name}.${System.nanoTime()}.tmp")
        tmp.writeText(content, Charsets.UTF_8)
        if (!tmp.renameTo(f)) { f.delete(); tmp.renameTo(f) }
        dao.updateWordCount(bookId, content.count { !it.isWhitespace() })
        dao.setHash(bookId, content.hashCode().toString(16))
    }

    private fun flushNow() {
        if (!dirty) return
        saveRunnable?.let { mainHandler.removeCallbacks(it) }
        val b = book ?: return
        val text = editorText?.text?.toString() ?: return
        dirty = false
        ioExecutor.execute { writeContent(b.id, text) }
    }

    private fun updateWordCount(text: String) {
        wordCountView?.text = "${text.count { !it.isWhitespace() }} 瀛?
    }
}

package com.example.novel.db

import android.content.ContentValues
import android.content.Context
import android.database.Cursor
import java.util.UUID

data class Annotation(
    val id: String,
    val bookId: String,
    val startOffset: Int,
    val endOffset: Int,
    val quotedText: String,
    val note: String? = null,
    val color: String = "yellow",
    val createdAt: Long
)

class AnnotationDao(context: Context) {
    private val helper = DbHelper(context.applicationContext)
    private fun db() = helper.writableDatabase

    fun insert(bookId: String, start: Int, end: Int, quotedText: String, note: String? = null, color: String = "yellow"): Annotation {
        val id = UUID.randomUUID().toString()
        val now = System.currentTimeMillis()
        val cv = ContentValues().apply {
            put("id", id)
            put("book_id", bookId)
            put("chapter_id", "main")
            put("start_offset", start)
            put("end_offset", end)
            put("color", color)
            put("style", "highlight")
            put("note", note)
            put("quoted_text", quotedText)
            put("orphan", 0)
            put("created_at", now)
            put("updated_at", now)
        }
        db().insert(DbHelper.T_ANNOTATION, null, cv)
        return Annotation(id, bookId, start, end, quotedText, note, color, now)
    }

    fun getAnnotations(bookId: String): List<Annotation> {
        val list = mutableListOf<Annotation>()
        db().query(
            DbHelper.T_ANNOTATION, null,
            "book_id=? AND orphan=0", arrayOf(bookId),
            null, null, "start_offset ASC"
        ).use { c ->
            while (c.moveToNext()) {
                list.add(
                    Annotation(
                        id = c.getString(c.getColumnIndexOrThrow("id")),
                        bookId = c.getString(c.getColumnIndexOrThrow("book_id")),
                        startOffset = c.getInt(c.getColumnIndexOrThrow("start_offset")),
                        endOffset = c.getInt(c.getColumnIndexOrThrow("end_offset")),
                        quotedText = c.getString(c.getColumnIndexOrThrow("quoted_text")),
                        note = c.getString(c.getColumnIndexOrThrow("note")),
                        color = c.getString(c.getColumnIndexOrThrow("color")),
                        createdAt = c.getLong(c.getColumnIndexOrThrow("created_at"))
                    )
                )
            }
        }
        return list
    }

    fun delete(id: String) {
        db().delete(DbHelper.T_ANNOTATION, "id=?", arrayOf(id))
    }
}

package com.example.novel.db

import android.content.ContentValues
import android.content.Context
import android.database.Cursor
import android.database.sqlite.SQLiteDatabase

data class Book(
    val id: String,
    val title: String,
    val isShort: Boolean,
    val wordCount: Int,
    val updatedAt: Long,
    val contentHash: String?
)

class BookDao(context: Context) {
    private val helper = DbHelper(context.applicationContext)
    private fun db() = helper.writableDatabase

    fun addBook(title: String, isShort: Boolean): Book? {
        val t = title.trim()
        if (t.isEmpty()) return null
        val id = System.currentTimeMillis().toString()
        val now = System.currentTimeMillis()
        val ok = addBookWithId(id, t, isShort, 0, now, "")
        return if (ok) Book(id, t, isShort, 0, now, "") else null
    }

    fun addBookWithId(id: String, title: String, isShort: Boolean, wordCount: Int, updatedAt: Long, hash: String?): Boolean {
        if (id.isBlank() || title.isBlank()) return false
        val d = db()
        if (exists(d, id)) return false
        val cv = ContentValues().apply {
            put("book_id", id)
            put("title", title)
            put("is_short", if (isShort) 1 else 0)
            put("content_hash", hash ?: "")
            put("word_count", wordCount)
            put("updated_at", updatedAt)
        }
        return d.insert(DbHelper.T_BOOK_META, null, cv) != -1L
    }

    fun getAllBooks(isShort: Boolean): List<Book> {
        val list = mutableListOf<Book>()
        db().query(DbHelper.T_BOOK_META, null, "is_short=?", arrayOf(if (isShort) "1" else "0"), null, null, "updated_at DESC").use { c -> 
            while (c.moveToNext()) list.add(readBook(c)) 
        }
        return list
    }

    fun getBook(id: String): Book? {
        db().query(DbHelper.T_BOOK_META, null, "book_id=?", arrayOf(id), null, null, null).use { c -> 
            if (c.moveToFirst()) return readBook(c) 
        }
        return null
    }

    fun deleteBook(id: String) {
        val d = db()
        d.delete(DbHelper.T_BOOK_META, "book_id=?", arrayOf(id))
        d.delete(DbHelper.T_ANNOTATION, "book_id=?", arrayOf(id))
        d.delete(DbHelper.T_SESSION, "book_id=?", arrayOf(id))
    }

    fun updateWordCount(id: String, wordCount: Int) {
        val cv = ContentValues().apply {
            put("word_count", wordCount)
            put("updated_at", System.currentTimeMillis())
        }
        db().update(DbHelper.T_BOOK_META, cv, "book_id=?", arrayOf(id))
    }

    fun setHash(id: String, hash: String) {
        val cv = ContentValues().apply {
            put("content_hash", hash)
            put("updated_at", System.currentTimeMillis())
        }
        db().update(DbHelper.T_BOOK_META, cv, "book_id=?", arrayOf(id))
    }

    fun isEmpty(): Boolean {
        db().query(DbHelper.T_BOOK_META, arrayOf("book_id"), null, null, null, null, null).use { c -> return !c.moveToFirst() }
    }

    private fun exists(d: SQLiteDatabase, id: String): Boolean {
        d.query(DbHelper.T_BOOK_META, arrayOf("book_id"), "book_id=?", arrayOf(id), null, null, null).use { c -> return c.moveToFirst() }
    }

    private fun readBook(c: Cursor): Book = Book(
        id = c.getString(c.getColumnIndexOrThrow("book_id")),
        title = c.getString(c.getColumnIndexOrThrow("title")),
        isShort = c.getInt(c.getColumnIndexOrThrow("is_short")) == 1,
        wordCount = c.getInt(c.getColumnIndexOrThrow("word_count")),
        updatedAt = c.getLong(c.getColumnIndexOrThrow("updated_at")),
        contentHash = c.getString(c.getColumnIndexOrThrow("content_hash"))
    )
}

package com.example.novel.db

import android.content.Context
import android.util.Log
import org.json.JSONArray
import java.io.File

object JsonMigrator {
    private const val TAG = "JsonMigrator"
    private val CANDIDATES = listOf("novel_data.json", "books.json")

    fun migrateIfNeeded(context: Context) {
        val dir = context.filesDir
        val oldFile = CANDIDATES.map { File(dir, it) }.firstOrNull { it.exists() }
        if (oldFile == null) return

        val dao = BookDao(context)
        if (!dao.isEmpty()) return

        try {
            val text = oldFile.readText(Charsets.UTF_8).trim()
            if (text.isEmpty()) return
            val arr = JSONArray(text)
            for (i in 0 until arr.length()) {
                val o = arr.optJSONObject(i) ?: continue
                val id = o.optString("id").trim()
                val title = o.optString("title").trim()
                if (id.isEmpty() || title.isEmpty()) continue
                val isShort = (o.optString("type").trim() == "鐭瘒")
                val content = o.optString("content")
                val wordCount = content.replace("\\s".toRegex(), "").length
                val hash = content.hashCode().toString(16)
                val updatedAt = id.toLongOrNull() ?: System.currentTimeMillis()
                dao.addBookWithId(id, title, isShort, wordCount, updatedAt, hash)
            }
        } catch (e: Exception) {
            Log.e(TAG, "杩佺Щ澶辫触", e)
        }
    }
}
